# Модуль 2.1. C++ для MCU: Обмеження ресурсів та особливості

## Опис завдання

### Концепції Embedded C++:

- volatile, const / constexpr, static
- Відмова від динамічної пам’яті та важких контейнерів STL
- Superloop-архітектура
- Безпечна робота з периферією (LED, кнопка)

### 1. Переписати “класичний Arduino blink” у Embedded C++

- Використати enum class для стану LED (On / Off).
- Використати constexpr для номера піну та інтервалу блимання.
- Уникнути глобальних змінних (крім мінімально необхідних для стану/ISR).
- Не використовувати delay(); реалізувати неблокуючий код у superloop за допомогою millis().
- Створити клас Led з методами init() та set(LedState state).

### 2. Використати static const / constexpr для параметрів

- Створити клас-конфігурацію або структуру з static const / constexpr для додаткових налаштувань.
- Повністю усунути «магічні числа», винісши їх у константи.

### 3. Виміряти час виконання superloop (Опційно)

- Виміряти час однієї ітерації loop() без delay().
- Виводити середній час або показники у Serial Monitor кожні 1000 ітерацій.

### 4. Додати кнопку з обробкою через переривання (опційно)

- Під'єднати кнопку до GPIO та налаштувати переривання (attachInterrupt)
- Використати volatile bool button прапорець для фіксації події натискання в ISR.
- Змінювати режим роботи LED при кожному натисканні: Блимання → Постійно увімкнено → Постійно вимкнено → Блимання.
- Зробити ISR мінімальним (без викликів Serial.print чи затримок). 
- Всю логіку перемикання режимів та антидребезгу (debouncing) обробляти в loop().

## Код завдання

```cpp
#include <Arduino.h>

// --- 2. Конфігурація параметрів (без "магічних чисел") ---
struct Config {
    static constexpr uint8_t LedPin = 2;               // Пін підключення світлодіода
    static constexpr uint8_t ButtonPin = 0;            // Пін кнопки (BOOT кнопка на ESP32)
    static constexpr uint32_t BlinkIntervalMs = 500;   // Інтервал перемикання LED (мс)
    static constexpr uint32_t DebounceDelayMs = 200;   // Захист від брязкоту контактів (мс)
    static constexpr uint32_t LoopMeasureCount = 1000; // Кількість ітерацій для вимірювання
    static constexpr uint32_t SerialBaudRate = 115200; // Швидкість Serial-порту
};

// --- 1. Безпечні перелічення (enum class) ---
enum class LedState : uint8_t {
    Off = LOW,
    On = HIGH
};

enum class LedMode : uint8_t {
    Blinking = 0,
    AlwaysOn,
    AlwaysOff
};

// --- 1. Клас для інкапсуляції роботи з LED ---
class Led {
private:
    const uint8_t pin;
    LedState state{LedState::Off};

public:
    explicit constexpr Led(uint8_t pin) : pin(pin) {}

    void init() {
        pinMode(pin, OUTPUT);
        set(LedState::Off);
    }

    void set(LedState newState) {
        state = newState;
        digitalWrite(pin, static_cast<uint8_t>(state));
    }

    void toggle() {
        set(state == LedState::On ? LedState::Off : LedState::On);
    }

    LedState getState() const {
        return state;
    }
};

// Єдиний об'єкт периферії на рівні модуля та мінімальний прапорець для ISR
static Led led(Config::LedPin);
static volatile bool buttonPressed = false; // volatile гарантує коректне читання значення, зміненого в ISR

// --- 4. Мінімальний обробник переривання (ISR) ---
void IRAM_ATTR buttonIsr() {
    buttonPressed = true; // Тільки фіксація події; без викликів функцій та затримок
}

void setup() {
    Serial.begin(Config::SerialBaudRate);
    delay(1000); // Час на встановлення зв'язку з Serial Monitor після ресету

    // Ініціалізація світлодіода
    led.init();

    // Налаштування кнопки з внутрішньою підтяжкою та перериванням по спаду сигналу
    pinMode(Config::ButtonPin, INPUT_PULLUP);
    attachInterrupt(digitalPinToInterrupt(Config::ButtonPin), buttonIsr, FALLING);
}

void loop() {
    // 3. Замір часу початку ітерації
    const uint32_t loopStartTimeUs = micros();

    // Статичні змінні зберігають стан між викликами loop, уникаючи глобальної області видимості
    static LedMode currentMode = LedMode::Blinking;
    static uint32_t lastBlinkTimeMs = 0;
    static uint32_t lastButtonPressMs = 0;
    static uint32_t loopCounter = 0;
    static uint64_t accumulatedLoopDurationUs = 0;

    const uint32_t currentMillis = millis();

    // --- 4. Обробка прапорця переривання та програмний антидребезг (debouncing) ---
    if (buttonPressed) {
        buttonPressed = false;
        if (currentMillis - lastButtonPressMs >= Config::DebounceDelayMs) {
            lastButtonPressMs = currentMillis;

            // Циклічне перемикання: Blinking -> AlwaysOn -> AlwaysOff -> Blinking
            switch (currentMode) {
                case LedMode::Blinking:
                    currentMode = LedMode::AlwaysOn;
                    Serial.println("[MODE] Always On");
                    break;
                case LedMode::AlwaysOn:
                    currentMode = LedMode::AlwaysOff;
                    Serial.println("[MODE] Always Off");
                    break;
                case LedMode::AlwaysOff:
                    currentMode = LedMode::Blinking;
                    Serial.println("[MODE] Blinking");
                    break;
            }
        }
    }

    // --- 1 & 4. Неблокуюче керування LED (Superloop) ---
    switch (currentMode) {
        case LedMode::Blinking:
            if (currentMillis - lastBlinkTimeMs >= Config::BlinkIntervalMs) {
                lastBlinkTimeMs = currentMillis;
                led.toggle();
            }
            break;

        case LedMode::AlwaysOn:
            led.set(LedState::On);
            break;

        case LedMode::AlwaysOff:
            led.set(LedState::Off);
            break;
    }

    // --- 3. Вимірювання середнього часу виконання loop() ---
    accumulatedLoopDurationUs += (micros() - loopStartTimeUs);
    ++loopCounter;

    if (loopCounter >= Config::LoopMeasureCount) {
        const float avgLoopTimeUs = static_cast<float>(accumulatedLoopDurationUs) / Config::LoopMeasureCount;
        Serial.print("Avg loop time: ");
        Serial.print(avgLoopTimeUs);
        Serial.println(" us");

        loopCounter = 0;
        accumulatedLoopDurationUs = 0;
    }
}
```
